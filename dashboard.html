// gcal-create-event // Called by an agent's browser session when a booking is created. Pushes a // matching event onto the venue's and/or band's connected Google Calendar, // if they have one connected. Silently does nothing for an owner with no // connection — not every venue/band will have connected their calendar. // // Expects a normal authenticated call (Supabase JWT verification stays ON // for this function) since only a logged-in agent should be able to // trigger this, and only for venues/bands in their own business. import { createClient } from 'jsr:@supabase/supabase-js@2'; const GOOGLE_CLIENT_ID = Deno.env.get('GOOGLE_CLIENT_ID')!; const GOOGLE_CLIENT_SECRET = Deno.env.get('GOOGLE_CLIENT_SECRET')!; const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!; const SERVICE_ROLE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!; const DEFAULT_TIMEZONE = 'America/Chicago'; // TODO: make this configurable per business later async function refreshAccessToken(refreshToken: string): Promise<{ accessToken: string; expiresAt: string } | null> { const res = await fetch('https://oauth2.googleapis.com/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ refresh_token: refreshToken, client_id: GOOGLE_CLIENT_ID, client_secret: GOOGLE_CLIENT_SECRET, grant_type: 'refresh_token' }) }); const data = await res.json(); if (!res.ok || !data.access_token) { console.error('Refresh failed', data); return null; } const expiresAt = new Date(Date.now() + (data.expires_in || 3600) * 1000).toISOString(); return { accessToken: data.access_token, expiresAt }; } async function pushEventForOwner( supabase: any, ownerType: string, ownerId: string, eventPayload: any ): Promise<{ pushed: boolean; reason?: string }> { const connResult = await supabase .from('calendar_connections') .select('*') .eq('owner_type', ownerType) .eq('owner_id', ownerId) .maybeSingle(); if (connResult.error || !connResult.data) { return { pushed: false, reason: 'not_connected' }; } let conn = connResult.data; let accessToken = conn.access_token; const expiresAt = new Date(conn.token_expiry).getTime(); const needsRefresh = expiresAt < Date.now() + 60 * 1000; // refresh if expired or about to expire if (needsRefresh) { const refreshed = await refreshAccessToken(conn.refresh_token); if (!refreshed) return { pushed: false, reason: 'refresh_failed' }; accessToken = refreshed.accessToken; await supabase.from('calendar_connections').update({ access_token: refreshed.accessToken, token_expiry: refreshed.expiresAt, updated_at: new Date().toISOString() }).eq('owner_type', ownerType).eq('owner_id', ownerId); } const createRes = await fetch('https://www.googleapis.com/calendar/v3/calendars/primary/events', { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + accessToken }, body: JSON.stringify(eventPayload) }); if (!createRes.ok) { const errBody = await createRes.text(); console.error('Calendar event creation failed', errBody); return { pushed: false, reason: 'google_api_error' }; } return { pushed: true }; } const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', }; Deno.serve(async (req) => { if (req.method === 'OPTIONS') return new Response('ok', { headers: cors }); if (req.method !== 'POST') { return new Response('Method not allowed', { status: 405, headers: cors }); } const authHeader = req.headers.get('Authorization') || ''; const jwt = authHeader.replace('Bearer ', ''); const userClient = createClient(SUPABASE_URL, Deno.env.get('SUPABASE_ANON_KEY')!, { global: { headers: { Authorization: authHeader } } }); const userResult = await userClient.auth.getUser(jwt); if (userResult.error || !userResult.data.user) { return new Response(JSON.stringify({ error: 'Not authenticated' }), { status: 401, headers: { ...cors, 'Content-Type': 'application/json' } }); } const body = await req.json(); const venueId = body.venueId || null; const bandId = body.bandId || null; const venueName = body.venueName || ''; const bandName = body.bandName || ''; const date = body.date; // 'YYYY-MM-DD' const startTime = body.startTime; // 'HH:MM' const endTime = body.endTime; // 'HH:MM' const notes = body.notes || ''; if (!date || !startTime || !endTime || (!venueId && !bandId)) { return new Response(JSON.stringify({ error: 'Missing required booking fields' }), { status: 400, headers: { ...cors, 'Content-Type': 'application/json' } }); } const admin = createClient(SUPABASE_URL, SERVICE_ROLE_KEY); // Confirm the calling agent's business actually owns this venue/band before touching any calendar const profileResult = await admin.from('profiles').select('business_id').eq('id', userResult.data.user.id).single(); if (profileResult.error || !profileResult.data) { return new Response(JSON.stringify({ error: 'Agent profile not found' }), { status: 403, headers: { ...cors, 'Content-Type': 'application/json' } }); } const callerBusinessId = profileResult.data.business_id; if (venueId) { const v = await admin.from('venues').select('business_id').eq('id', venueId).single(); if (v.error || !v.data || v.data.business_id !== callerBusinessId) { return new Response(JSON.stringify({ error: 'Venue not found in your business' }), { status: 403, headers: { ...cors, 'Content-Type': 'application/json' } }); } } if (bandId) { const b = await admin.from('bands').select('business_id').eq('id', bandId).single(); if (b.error || !b.data || b.data.business_id !== callerBusinessId) { return new Response(JSON.stringify({ error: 'Band not found in your business' }), { status: 403, headers: { ...cors, 'Content-Type': 'application/json' } }); } } const startDateTime = date + 'T' + startTime + ':00'; const endDateTime = date + 'T' + endTime + ':00'; const results: Record = {}; if (venueId) { const eventPayload = { summary: 'Gig: ' + bandName, description: 'Booked via Gig Matrix.' + (notes ? '\n\nNotes: ' + notes : ''), start: { dateTime: startDateTime, timeZone: DEFAULT_TIMEZONE }, end: { dateTime: endDateTime, timeZone: DEFAULT_TIMEZONE } }; results.venue = await pushEventForOwner(admin, 'venue', venueId, eventPayload); } if (bandId) { const eventPayload = { summary: 'Gig at ' + venueName, description: 'Booked via Gig Matrix.' + (notes ? '\n\nNotes: ' + notes : ''), start: { dateTime: startDateTime, timeZone: DEFAULT_TIMEZONE }, end: { dateTime: endDateTime, timeZone: DEFAULT_TIMEZONE } }; results.band = await pushEventForOwner(admin, 'band', bandId, eventPayload); } return new Response(JSON.stringify({ success: true, results }), { headers: { ...cors, 'Content-Type': 'application/json' } }); });
